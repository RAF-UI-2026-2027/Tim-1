Enum account_status {
  active
  suspended
  banned
  deleted
}

Enum friendship_status {
  pending
  accepted
  blocked
  declined
}

Enum penalty_type {
  warning
  chat_restriction
  ranked_restriction
  temporary_ban
  permanent_ban
}

Table regions {
  region_id int [pk, increment]
  code varchar(10) [unique, not null, note: 'EUW, EUNE, NA, KR...']
  name varchar(50) [not null]
  server_location varchar(100)
  timezone varchar(50)
  is_active boolean [not null]
}

Table games {
  game_id int [pk, increment]
  name varchar(100) [unique, not null, note: 'League of Legends, Valorant...']
  short_code varchar(10) [unique, not null, note: 'LOL, VAL, TFT, LOR']
  genre varchar(50)
  release_date date
  is_active boolean [not null]
}

Table accounts {
  account_id bigint [pk, increment]
  puuid uuid [unique, not null, note: 'Globalni jedinstveni ID igraca']
  game_name varchar(16) [not null, note: 'Deo Riot ID-a pre #']
  tag_line varchar(5) [not null, note: 'Deo Riot ID-a posle #']
  email varchar(255) [unique, not null]
  password varchar(255) [not null]
  date_of_birth date [not null]
  country_code char(2)
  home_region_id int [not null, ref: > regions.region_id]
  status account_status [not null]
  created_at timestamp [not null]
  last_login_at timestamp
}

Table game_accounts {
  game_account_id bigint [pk, increment]
  account_id bigint [not null, ref: > accounts.account_id]
  game_id int [not null, ref: > games.game_id]
  region_id int [not null, ref: > regions.region_id]
  account_level int [not null, default: 1]
  first_played_at timestamp
  last_played_at timestamp
}

Table login_sessions {
  session_id bigint [pk, increment]
  account_id bigint [not null, ref: > accounts.account_id]
  ip_address varchar(45) [not null]
  device_type varchar(50)
  client_version varchar(20)
  is_successful boolean [not null]
  login_at timestamp [not null]
  logout_at timestamp
}

Table friendships {
  friendship_id bigint [pk, increment]
  requester_id bigint [not null, ref: > accounts.account_id]
  addressee_id bigint [not null, ref: > accounts.account_id]
  status friendship_status [not null, default: 'pending']
  created_at timestamp [not null]
  responded_at timestamp
}

Table account_penalties {
  penalty_id bigint [pk, increment]
  account_id bigint [not null, ref: > accounts.account_id]
  game_id int [ref: > games.game_id, not null]
  type penalty_type [not null]
  reason text [not null]
  issued_at timestamp [not null]
  expires_at timestamp
  is_active boolean [not null]
}
