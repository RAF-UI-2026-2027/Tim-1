Table regions {
  region_id int [pk, increment]
  code varchar(10) [unique, not null]
  name varchar(50) [not null]
  server_location varchar(100)
  timezone varchar(50)
  is_active boolean [not null]
}

Table games {
  game_id int [pk, increment]
  name varchar(100) [unique, not null]
  short_code varchar(10) [unique, not null]
  genre varchar(50)
  release_date date
  is_active boolean [not null]
}

Table accounts {
  account_id bigint [pk, increment]
  puuid uuid [unique, not null]
  game_name varchar(16) [not null]
  tag_line varchar(5) [not null]
  email varchar(255) [unique, not null]
  password_hash varchar(255) [not null]
  date_of_birth date [not null]
  country varchar(40)
  home_region_id int [not null, ref: > regions.region_id]
  created_at timestamp [not null]
}

Table account_statuses {
  status_id int [pk, increment]
  name varchar(30) [unique, not null]
  description text
}

Table account_status_history {
  account_status_history_id bigint [pk, increment]
  account_id bigint [not null, ref: > accounts.account_id]
  status_id int [not null, ref: > account_statuses.status_id]
  valid_from timestamp [not null]
  valid_to timestamp
  reason text
}

Table game_accounts {
  game_account_id bigint [pk, increment]
  account_id bigint [not null, ref: > accounts.account_id]
  game_id int [not null, ref: > games.game_id]
  region_id int [not null, ref: > regions.region_id]
  account_level int [not null, default: 1]
  first_played_at timestamp
  last_played_at timestamp
}

Table launcher_sessions {
  launcher_session_id bigint [pk, increment]
  account_id bigint [not null, ref: > accounts.account_id]
  ip_address varchar(45) [not null]
  device_type varchar(50)
  operating_system varchar(50)
  launcher_version varchar(20)
  is_successful boolean [not null]
  login_at timestamp [not null]
  logout_at timestamp
}

Table game_sessions {
  game_session_id bigint [pk, increment]
  launcher_session_id bigint [not null, ref: > launcher_sessions.launcher_session_id]
  game_account_id bigint [not null, ref: > game_accounts.game_account_id]
  client_version varchar(20)
  started_at timestamp [not null]
  ended_at timestamp
}

Table friend_requests {
  friend_request_id bigint [pk, increment]
  sender_id bigint [not null, ref: > accounts.account_id]
  receiver_id bigint [not null, ref: > accounts.account_id]
  sent_at timestamp [not null]
}

Table friend_request_statuses {
  request_status_id int [pk, increment]
  name varchar(30) [unique, not null]
  description text
}

Table friend_request_status_history {
  request_status_history_id bigint [pk, increment]
  friend_request_id bigint [not null, ref: > friend_requests.friend_request_id]
  request_status_id int [not null, ref: > friend_request_statuses.request_status_id]
  valid_from timestamp [not null]
  valid_to timestamp
}

Table friendships {
  friendship_id bigint [pk, increment]
  account1_id bigint [not null, ref: > accounts.account_id]
  account2_id bigint [not null, ref: > accounts.account_id]
  friend_request_id bigint [unique, not null, ref: - friend_requests.friend_request_id]
  friends_since timestamp [not null]
  ended_at timestamp
}

Table blocked_players {
  block_id bigint [pk, increment]
  blocker_id bigint [not null, ref: > accounts.account_id]
  blocked_id bigint [not null, ref: > accounts.account_id]
  reason varchar(255)
  blocked_at timestamp [not null]
  unblocked_at timestamp
}

Table penalty_types {
  penalty_type_id int [pk, increment]
  name varchar(50) [unique, not null]
  description text
  severity int [not null]
}

Table account_penalties {
  penalty_id bigint [pk, increment]
  account_id bigint [not null, ref: > accounts.account_id]
  game_id int [not null, ref: > games.game_id]
  reason text [not null]
  issued_at timestamp [not null]
}

Table account_penalty_types {
  account_penalty_type_id bigint [pk, increment]
  penalty_id bigint [not null, ref: > account_penalties.penalty_id]
  penalty_type_id int [not null, ref: > penalty_types.penalty_type_id]
  valid_from timestamp [not null]
  valid_to timestamp
}

Table riot_id_history {
  riot_id_history_id bigint [pk, increment]
  account_id bigint [not null, ref: > accounts.account_id]
  old_game_name varchar(16) [not null]
  old_tag_line varchar(5) [not null]
  new_game_name varchar(16) [not null]
  new_tag_line varchar(5) [not null]
  changed_at timestamp [not null]
}

Table lol_client_settings {
  client_settings_id bigint [pk, increment]
  game_account_id bigint [unique, not null, ref: - game_accounts.game_account_id]
  language varchar(10) [not null]
  window_size varchar(20)
  low_spec_mode boolean [not null]
  close_client_during_game boolean [not null]
  enable_sounds boolean [not null]
  updated_at timestamp [not null]
}

Table queue_types {
  queue_type_id int [pk, increment]
  game_id int [not null, ref: > games.game_id]
  name varchar(50) [unique, not null]
  map_name varchar(50) [not null]
  team_size int [not null]
  max_party_size int [not null]
  is_ranked boolean [not null]
  is_active boolean [not null]
}

Table positions {
  position_id int [pk, increment]
  name varchar(30) [unique, not null]
  short_name varchar(10) [unique, not null]
  description text
}

Table lobbies {
  lobby_id bigint [pk, increment]
  queue_type_id int [not null, ref: > queue_types.queue_type_id]
  leader_game_account_id bigint [not null, ref: > game_accounts.game_account_id]
  is_private boolean [not null]
  created_at timestamp [not null]
  closed_at timestamp
}

Table lobby_members {
  lobby_member_id bigint [pk, increment]
  lobby_id bigint [not null, ref: > lobbies.lobby_id]
  game_account_id bigint [not null, ref: > game_accounts.game_account_id]
  joined_at timestamp [not null]
  left_at timestamp
}

Table lobby_member_positions {
  lobby_member_position_id bigint [pk, increment]
  lobby_member_id bigint [not null, ref: > lobby_members.lobby_member_id]
  position_id int [not null, ref: > positions.position_id]
  preference_order int [not null]
}

Table matchmaking_tickets {
  ticket_id bigint [pk, increment]
  lobby_id bigint [not null, ref: > lobbies.lobby_id]
  average_mmr int [not null]
  estimated_wait_seconds int
  queued_at timestamp [not null]
  left_queue_at timestamp
}
